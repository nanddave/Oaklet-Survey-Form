Integration Code

Add this simple script tag to your website and start collecting patient data:

<!-- Add to your HTML head --> <script src="https://survey.oaklet.com/widget.js"></script> <!-- Add this button anywhere on your page --> <button onclick="OakletSurvey.open('your-org-id')"> Schedule Consultation </button> <!-- Or create a custom button --> <script> const button = OakletSurvey.createButton({ organizationId: 'your-org-id', buttonText: 'Book Appointment', theme: { primaryColor: '#059669', borderRadius: '8px' }, onComplete: (result) => { console.log('Survey completed:', result); // Handle successful completion }, onError: (error) => { console.error('Survey error:', error); // Handle errors } }); document.getElementById('schedule-button').appendChild(button); </script>
